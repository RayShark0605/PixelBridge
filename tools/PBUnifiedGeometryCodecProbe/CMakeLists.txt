cmake_minimum_required(VERSION 3.24)
project(PBUnifiedGeometryCodecProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(PB_GEOM_REPO "" CACHE PATH "Explicit checkout")
set(PB_GEOM_BASE_BUILD "" CACHE PATH "Read-only Step2 Release build")
set(PB_GEOM_MEDIA "" CACHE PATH "Read-only Step2 media dependency")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${PB_GEOM_BASE_BUILD}/libs/PBModulation/Release/PBModulation.lib" OR NOT EXISTS "${PB_GEOM_MEDIA}/include/libavcodec/avcodec.h")
    message(FATAL_ERROR "Explicit sealed Windows/MSVC dependencies required; no installation")
endif()
execute_process(COMMAND "<python>" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_reference.py" --repo "${PB_GEOM_REPO}" --output "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
set(dependencies "${PB_GEOM_BASE_BUILD}/vcpkg_installed/x64-windows")
add_executable(PBUnifiedGeometryCodecProbe main.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/locator_probe.cpp"
    "${PB_GEOM_REPO}/tools/PBUnifiedRecordingReplay/recording_media.cpp")
target_compile_options(PBUnifiedGeometryCodecProbe PRIVATE /utf-8 /W4 /WX /EHsc /permissive- /external:anglebrackets /external:W0)
target_compile_definitions(PBUnifiedGeometryCodecProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(PBUnifiedGeometryCodecProbe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PB_GEOM_REPO}/libs/PBModulation/src"
    "${PB_GEOM_REPO}/tools/PBRemoteThroughputStep3B" "${PB_GEOM_REPO}/tools/PBUnifiedRecordingReplay" "${PB_GEOM_REPO}/apps/common")
target_include_directories(PBUnifiedGeometryCodecProbe SYSTEM PRIVATE "${dependencies}/include" "${PB_GEOM_MEDIA}/include")
foreach(library PBModulation PBCore PBProtocol PBInnerFec PBInterleave)
    target_include_directories(PBUnifiedGeometryCodecProbe PRIVATE "${PB_GEOM_REPO}/libs/${library}/include")
    target_link_libraries(PBUnifiedGeometryCodecProbe PRIVATE "${PB_GEOM_BASE_BUILD}/libs/${library}/Release/${library}.lib")
endforeach()
foreach(library blake3 libpng16 z)
    target_link_libraries(PBUnifiedGeometryCodecProbe PRIVATE "${dependencies}/lib/${library}.lib")
endforeach()
target_link_libraries(PBUnifiedGeometryCodecProbe PRIVATE bcrypt)
foreach(library avformat avcodec avutil swscale)
    target_link_libraries(PBUnifiedGeometryCodecProbe PRIVATE "${PB_GEOM_MEDIA}/lib/${library}.lib")
endforeach()
foreach(dll blake3.dll libpng16.dll z.dll)
    add_custom_command(TARGET PBUnifiedGeometryCodecProbe POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dependencies}/bin/${dll}" "$<TARGET_FILE_DIR:PBUnifiedGeometryCodecProbe>" VERBATIM)
endforeach()
foreach(dll avformat-62.dll avcodec-62.dll avutil-60.dll swscale-9.dll)
    add_custom_command(TARGET PBUnifiedGeometryCodecProbe POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${PB_GEOM_MEDIA}/bin/${dll}" "$<TARGET_FILE_DIR:PBUnifiedGeometryCodecProbe>" VERBATIM)
endforeach()
