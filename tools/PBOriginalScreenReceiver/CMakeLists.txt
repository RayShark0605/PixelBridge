cmake_minimum_required(VERSION 3.24)
project(PBOriginalScreenReceiver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(PB_STEP3B_REPO "" CACHE PATH "Explicit checkout")
set(PB_STEP3B_BASE_BUILD "" CACHE PATH "Existing original observation Release libraries")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${PB_STEP3B_BASE_BUILD}/apps/common/Release/PBApplication.lib")
    message(FATAL_ERROR "Explicit original Windows/MSVC observation build required")
endif()
set(dependencies "${PB_STEP3B_REPO}/build-remote-step2-20260908-run01/vcpkg_installed/x64-windows")
add_executable(PBOriginalScreenReceiver main.cpp "${PB_STEP3B_REPO}/cmake/PixelBridge.PerMonitorV2.manifest")
target_compile_options(PBOriginalScreenReceiver PRIVATE /utf-8 /W4 /WX /EHsc /permissive- /external:anglebrackets /external:W0)
target_compile_definitions(PBOriginalScreenReceiver PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(PBOriginalScreenReceiver PRIVATE "${PB_STEP3B_REPO}/apps/common" "${PB_STEP3B_REPO}/tools/PBRemoteThroughputStep3B")
target_include_directories(PBOriginalScreenReceiver SYSTEM PRIVATE "${dependencies}/include")
target_link_libraries(PBOriginalScreenReceiver PRIVATE "${PB_STEP3B_BASE_BUILD}/apps/common/Release/PBApplication.lib")
foreach(library PBReceiver PBCompression PBOuterFec PBStorage PBRenderD3D PBScreenCaptureWgc PBScreenCaptureDxgi PBDemodD3D11 PBTelemetry PBDesktopLevelsReference PBModulation PBCore PBInnerFec PBInterleave PBRealCaptureReplay PBCaptureNormalize PBScreenRegion PBPresentTiming PBProtocol)
    target_include_directories(PBOriginalScreenReceiver PRIVATE "${PB_STEP3B_REPO}/libs/${library}/include")
    target_link_libraries(PBOriginalScreenReceiver PRIVATE "${PB_STEP3B_BASE_BUILD}/libs/${library}/Release/${library}.lib")
endforeach()
foreach(library zstd wirehair libpng16 z blake3)
    target_link_libraries(PBOriginalScreenReceiver PRIVATE "${dependencies}/lib/${library}.lib")
endforeach()
target_link_libraries(PBOriginalScreenReceiver PRIVATE windowsapp wtsapi32 gdi32 imm32 d3dcompiler runtimeobject user32 bcrypt Shcore d3d11 dxgi)
foreach(dll blake3.dll zstd.dll)
    add_custom_command(TARGET PBOriginalScreenReceiver POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dependencies}/bin/${dll}" "$<TARGET_FILE_DIR:PBOriginalScreenReceiver>" VERBATIM)
endforeach()
target_link_options(PBOriginalScreenReceiver PRIVATE /MAP /VERBOSE:LIB)
