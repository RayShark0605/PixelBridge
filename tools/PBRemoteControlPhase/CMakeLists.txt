cmake_minimum_required(VERSION 3.24)
project(PBRemoteControlPhase LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(PB_PHASE_REPO "" CACHE PATH "Explicit source checkout")
set(PB_PHASE_BUILD "" CACHE PATH "Explicit sealed A or B Release build")
set(PB_PHASE_DEPENDENCIES "" CACHE PATH "Read-only installed x64-windows dependencies")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${PB_PHASE_BUILD}/apps/common/Release/PBApplication.lib")
    message(FATAL_ERROR "A separately built Windows/MSVC application is required")
endif()
add_library(PhaseDependencies INTERFACE)
target_compile_options(PhaseDependencies INTERFACE /utf-8 /W4 /WX /EHsc /permissive- /external:anglebrackets /external:W0)
target_compile_definitions(PhaseDependencies INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(PhaseDependencies INTERFACE "${PB_PHASE_REPO}/apps/common" "${PB_PHASE_REPO}/tools/PBRemoteThroughputStep3B")
target_include_directories(PhaseDependencies SYSTEM INTERFACE "${PB_PHASE_DEPENDENCIES}/include")
target_link_libraries(PhaseDependencies INTERFACE "${PB_PHASE_BUILD}/apps/common/Release/PBApplication.lib")
foreach(library PBReceiver PBCompression PBOuterFec PBStorage PBRenderD3D PBScreenCaptureWgc PBScreenCaptureDxgi PBDemodD3D11 PBTelemetry PBDesktopLevelsReference PBModulation PBCore PBInnerFec PBInterleave PBRealCaptureReplay PBCaptureNormalize PBScreenRegion PBPresentTiming PBProtocol)
    target_include_directories(PhaseDependencies INTERFACE "${PB_PHASE_REPO}/libs/${library}/include")
    target_link_libraries(PhaseDependencies INTERFACE "${PB_PHASE_BUILD}/libs/${library}/Release/${library}.lib")
endforeach()
foreach(library zstd wirehair libpng16 z blake3)
    target_link_libraries(PhaseDependencies INTERFACE "${PB_PHASE_DEPENDENCIES}/lib/${library}.lib")
endforeach()
target_link_libraries(PhaseDependencies INTERFACE windowsapp wtsapi32 gdi32 imm32 d3dcompiler runtimeobject user32 bcrypt Shcore d3d11 dxgi)

# The exact existing raw pixel entry is relinked, not replaced by packet input.
add_executable(PBControlPhaseRawReplay "${PB_PHASE_REPO}/tools/PBReceiverDecisionProbe/main.cpp")
add_executable(PBControlPhaseStriping main.cpp)
foreach(target PBControlPhaseRawReplay PBControlPhaseStriping)
    target_link_libraries(${target} PRIVATE PhaseDependencies)
    set_target_properties(${target} PROPERTIES VS_GLOBAL_VcpkgEnabled false VS_GLOBAL_VcpkgApplocalDeps false)
    target_link_options(${target} PRIVATE "/MAP:$<TARGET_FILE_DIR:${target}>/${target}.map")
    foreach(dll blake3.dll zstd.dll)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${PB_PHASE_DEPENDENCIES}/bin/${dll}" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
    endforeach()
endforeach()
