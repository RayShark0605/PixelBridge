cmake_minimum_required(VERSION 3.24)
project(PBRemoteThroughputStep3B LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(PB_STEP3B_REPO "" CACHE PATH "Explicit checkout")
set(PB_STEP3B_BASE_BUILD "" CACHE PATH "Read-only sealed Step2 Release build")
set(PB_STEP3B_MEDIA_ROOT "" CACHE PATH "Read-only Step2 media dependency, never the encoding DLLs")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${PB_STEP3B_BASE_BUILD}/apps/common/Release/PBApplication.lib" OR NOT EXISTS "${PB_STEP3B_MEDIA_ROOT}/include/libavcodec/avcodec.h")
    message(FATAL_ERROR "Explicit existing Windows/MSVC Step2 build and media dependency required")
endif()
set(dependencies "${PB_STEP3B_BASE_BUILD}/vcpkg_installed/x64-windows")
add_executable(PBRemoteThroughputStep3B main.cpp fixture.cpp
    "${PB_STEP3B_REPO}/tools/PBUnifiedRecordingReplay/recording_media.cpp")
target_compile_options(PBRemoteThroughputStep3B PRIVATE /utf-8 /W4 /WX /EHsc /permissive- /external:anglebrackets /external:W0)
target_compile_definitions(PBRemoteThroughputStep3B PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(PBRemoteThroughputStep3B PRIVATE "${PB_STEP3B_REPO}/apps/common" "${PB_STEP3B_REPO}/tools/PBUnifiedRecordingReplay")
target_include_directories(PBRemoteThroughputStep3B SYSTEM PRIVATE "${dependencies}/include" "${PB_STEP3B_MEDIA_ROOT}/include")
target_link_libraries(PBRemoteThroughputStep3B PRIVATE "${PB_STEP3B_BASE_BUILD}/apps/common/Release/PBApplication.lib")
foreach(library PBReceiver PBCompression PBOuterFec PBStorage PBRenderD3D PBScreenCaptureWgc PBScreenCaptureDxgi PBDemodD3D11 PBTelemetry PBDesktopLevelsReference PBModulation PBCore PBInnerFec PBInterleave PBRealCaptureReplay PBCaptureNormalize PBScreenRegion PBPresentTiming PBProtocol)
    target_include_directories(PBRemoteThroughputStep3B PRIVATE "${PB_STEP3B_REPO}/libs/${library}/include")
    target_link_libraries(PBRemoteThroughputStep3B PRIVATE "${PB_STEP3B_BASE_BUILD}/libs/${library}/Release/${library}.lib")
endforeach()
foreach(library zstd wirehair libpng16 z blake3)
    target_link_libraries(PBRemoteThroughputStep3B PRIVATE "${dependencies}/lib/${library}.lib")
endforeach()
foreach(library avformat avcodec avutil swscale)
    target_link_libraries(PBRemoteThroughputStep3B PRIVATE "${PB_STEP3B_MEDIA_ROOT}/lib/${library}.lib")
endforeach()
target_link_libraries(PBRemoteThroughputStep3B PRIVATE windowsapp wtsapi32 gdi32 imm32 d3dcompiler runtimeobject user32 bcrypt Shcore d3d11 dxgi)
foreach(dll blake3.dll zstd.dll)
    add_custom_command(TARGET PBRemoteThroughputStep3B POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dependencies}/bin/${dll}" "$<TARGET_FILE_DIR:PBRemoteThroughputStep3B>" VERBATIM)
endforeach()
foreach(dll avformat-62.dll avcodec-62.dll avutil-60.dll swscale-9.dll)
    add_custom_command(TARGET PBRemoteThroughputStep3B POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${PB_STEP3B_MEDIA_ROOT}/bin/${dll}" "$<TARGET_FILE_DIR:PBRemoteThroughputStep3B>" VERBATIM)
endforeach()
