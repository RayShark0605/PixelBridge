cmake_minimum_required(VERSION 3.24)
project(PBReceiverDecisionProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(PB_DECISION_REPO "" CACHE PATH "Explicit checkout")
set(PB_DECISION_BASE_BUILD "" CACHE PATH "Read-only Step2 Release dependencies")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${PB_DECISION_BASE_BUILD}/libs/PBReceiver/Release/PBReceiver.lib")
    message(FATAL_ERROR "Explicit existing Windows/MSVC Step2 dependencies required; no installation")
endif()
set(dependencies "${PB_DECISION_BASE_BUILD}/vcpkg_installed/x64-windows")
add_library(DecisionDependencies INTERFACE)
target_compile_options(DecisionDependencies INTERFACE /utf-8 /W4 /WX /EHsc /permissive- /external:anglebrackets /external:W0)
target_compile_definitions(DecisionDependencies INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
    PB_STEP1_SOURCE_FINGERPRINT="receiver-decision-experiment" PB_STEP1_BASE_COMMIT="4a36d0f1b84c94c7388a58b8249a3f1b6ec0634a")
target_include_directories(DecisionDependencies INTERFACE "${PB_DECISION_REPO}/apps/common" "${PB_DECISION_REPO}/tools/PBRemoteThroughputStep3B")
target_include_directories(DecisionDependencies SYSTEM INTERFACE "${dependencies}/include")
foreach(library PBReceiver PBCompression PBOuterFec PBStorage PBRenderD3D PBScreenCaptureWgc PBScreenCaptureDxgi PBDemodD3D11 PBTelemetry PBDesktopLevelsReference PBModulation PBCore PBInnerFec PBInterleave PBRealCaptureReplay PBCaptureNormalize PBScreenRegion PBPresentTiming PBProtocol)
    target_include_directories(DecisionDependencies INTERFACE "${PB_DECISION_REPO}/libs/${library}/include")
    target_link_libraries(DecisionDependencies INTERFACE "${PB_DECISION_BASE_BUILD}/libs/${library}/Release/${library}.lib")
endforeach()
foreach(library zstd wirehair libpng16 z blake3)
    target_link_libraries(DecisionDependencies INTERFACE "${dependencies}/lib/${library}.lib")
endforeach()
target_link_libraries(DecisionDependencies INTERFACE windowsapp wtsapi32 gdi32 imm32 d3dcompiler runtimeobject user32 bcrypt Shcore d3d11 dxgi)

# Rebuild current application sources; never link the stale Step2 runtime object.
add_library(DecisionApplication STATIC)
foreach(source application_model decoder_capture_controller decoder_resume_store encoder_session_store encoder_monitor_catalog evidence_journal local_desktop_runtime optional_diagnostic_fanout monitor_catalog remote_visual_replay_recorder run_report run_measurement recorded_pixel_replay sender_carousel_scheduler)
    target_sources(DecisionApplication PRIVATE "${PB_DECISION_REPO}/apps/common/${source}.cpp")
endforeach()
target_link_libraries(DecisionApplication PUBLIC DecisionDependencies)
add_executable(PBReceiverDecisionProbe main.cpp)
target_link_libraries(PBReceiverDecisionProbe PRIVATE DecisionApplication)
# Test TU includes the same runtime to access its private synchronous boundary.
# No test-only entry point or synthetic payload input is added to the product.
add_executable(PBReceiverDecisionTests test_decisions.cpp)
target_link_libraries(PBReceiverDecisionTests PRIVATE DecisionApplication)
foreach(target PBReceiverDecisionProbe PBReceiverDecisionTests)
    target_link_options(${target} PRIVATE "/MAP:$<TARGET_FILE_DIR:${target}>/${target}.map")
    foreach(dll blake3.dll zstd.dll)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${dependencies}/bin/${dll}" "$<TARGET_FILE_DIR:${target}>" VERBATIM)
    endforeach()
endforeach()
foreach(target DecisionApplication PBReceiverDecisionProbe PBReceiverDecisionTests)
    set_target_properties(${target} PROPERTIES VS_GLOBAL_VcpkgEnabled false VS_GLOBAL_VcpkgApplocalDeps false)
endforeach()
